public class IT23280656Lab7Q2A {
    public static void main(String[] args) {

        int row = 1;

        while (row <= 4) {

            int column = 1;

            while (column <= 5) {
                System.out.print("$ ");
                column++;
            }

            System.out.println();
            row++;
        }
    }
}

public class IT23306240Lab7Q2B {
    public static void main(String[] args) {

        int row = 1;

        while (row <= 5) {

            System.out.print(row + " - ");

            int star = 1;

            while (star <= row) {
                System.out.print("* ");
                star++;
            }

            System.out.println();
            row++;
        }
    }
}

public class IT23280656Lab7Q2C {
    public static void main(String[] args) {

        int number = 5;

        while (number >= 1) {

            int count = 1;

            while (count <= number) {
                System.out.print(number);
                count++;
            }

            System.out.println();
            number--;
        }
    }
}
